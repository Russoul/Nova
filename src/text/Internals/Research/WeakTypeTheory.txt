ℕ-elim : (P : ℕ -> 𝕌) (z : P 0) (s : (n : ℕ) -> P n -> P (S n)) -> (n : ℕ) -> P n
ℕ-elim-0 P z s : ℕ-elim P z s 0 = z ∈ P 0
ℕ-elim-S P z s x : ℕ-elim P z s (S x) = s x (ℕ-elim P z s x) ∈ P (S x)
