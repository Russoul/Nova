////////// Preface //////////

This file contains a formalisation of Nova's type theory: Extensional Type Theory.
We have clear distinction between syntax and derivations. In general, a derivation can't be recovered from the syntax alone.


//////////// Type Theory /////////////

We have:
 * Empty type: 𝟘
 * Unit type: 𝟙
 * Natural numbers type: ℕ
 * Dependent product type: Π
 * Dependent sum type: Σ
 * Identity type with reflection: Id
 * A universe: 𝕌

///////////// Syntax ///////////////

Σ, ϴ, Ξ ::= ε | Σ E  // signature (ground and dependent)
E ::= type | A       // signature entry (ground and dependent)
e ::= A | a          // signature entry
Γ, Δ ::= ε | Γ A     // ctx
Δ ::= ε | A Δ        // telescope
A ::=                // type
t ::=                // element

/////// Judgement forms /////////

Σ₀ ≡ Σ₁ sig  // signature

Σ₀ ≡ Σ₁ sig
----------------------- // signature entry
Σ₀ ≡ Σ₁ ⊦ E₀ ≡ E₁ entry

Σ₀ ≡ Σ₁ ⊦ E₀ ≡ E₁ entry
-------------------------------- // signature entry instance
Σ₀ ≡ Σ₁ ⊦ e₀ ≡ e₁ : E₀ ≡ E₁

Σ₀ ≡ Σ₁ sig
ϴ₀ ≡ ϴ₁ sig
--------------------------- // signature substitution
σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁

Σ₀ ≡ Σ₁ sig
----------------------  // context
Σ₀ ≡ Σ₁ ⊦ Γ₀ ≡ Γ₁ ctx

Σ₀ ≡ Σ₁ ⊦ Γ₀ ≡ Γ₁ ctx
Σ₀ ≡ Σ₁ ⊦ Δ₀ ≡ Δ₁ ctx
------------------------------------- // context substitution
Σ₀ ≡ Σ₁ ⊦ σ₀ ≡ σ₁ : Γ₀ ≡ Γ₁ ⇒ Δ₀ ≡ Δ₁

Σ₀ ≡ Σ₁ ⊦ Γ₀ ≡ Γ₁ ctx
---------------------------------- // type
(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ A₀ ≡ A₁ type

Σ₀ ≡ Σ₁ ⊦ Γ₀ ≡ Γ₁ ctx
--------------------------------- // telescope
(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ Δ₀ ≡ Δ₁ tel

(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ A₀ ≡ A₁ type
--------------------------------------- // element
(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ a₀ ≡ a₁ : A₀ ≡ A₁

(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ Δ₀ ≡ Δ₁ tel
--------------------------------------- // telescope instance
(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ ē₀ ≡ ē₁ : Δ₀ ≡ Δ₁

Σ₀ ≡ Σ₁ sig
------------------------ // dependent signature
(Σ₀ ≡ Σ₁) ⊦ ϴ₀ ≡ ϴ₁ dsig

Σ₀ ≡ Σ₁ ⊦ ϴ₀ ≡ ϴ₁ dsig
Σ₀ ≡ Σ₁ ⊦ Ξ₀ ≡ Ξ₁ dsig
------------------------------------- // dependent signature substitution
Σ₀ ≡ Σ₁ ⊦ σ₀ ≡ σ₁ : ϴ₀ ≡ ϴ₁ ⇒ Ξ₀ ≡ Ξ₁

(Σ₀ ≡ Σ₁) ⊦ ϴ₀ ≡ ϴ₁ dsig
----------------------------- // dependent signature instance
(Σ₀ ≡ Σ₁) ⊦ ē₀ ≡ ē₁ : ϴ₀ ≡ ϴ₁


///////// Inference rules (sig) //////////

ε ≡ ε sig                // empty signature

Σ₀ ≡ Σ₁ ⊦ E₀ ≡ E₁ entry
-----------------------  // signature extended by an entry
Σ₀ E₀ ≡ Σ₁ E₁ sig

Σ₀ ≡ Σ₁ ⊦ E₀ = E₁ entry
Σ₁ ≡ Σ₂ ⊦ E₁ = E₂ entry
-----------------------  // transitivity
Σ₀ ≡ Σ₂ ⊦ E₀ = E₂ entry

Σ₀ ≡ Σ₁ sig
Σ₀ ≡ Σ₁ ⊦ Ξ₀ ≡ Σ₁ dsig
----------------------------- // signature extended by a dependent signature (= multiple dependent entries)
(Σ₀ Ξ₀ ≡ Σ₁ Ξ₁) sig, where

  Σ ε = Σ               // by induction on syntax of dependent signature
  Σ (E Ξ) = (Σ E) Ξ

///////// Inference rules (sig entry) //////////

Σ₀ ≡ Σ₁ ⊦ Γ₀ ≡ Γ₁ ctx
-----------------------------------------  // Context extended by an type variable
Σ₀ ≡ Σ₁ ⊦ (Γ₀ ⊦ type) ≡ (Γ₁ ⊦ type) entry

(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ A₀ ≡ A₁ type
---------------------------------------    // Context extended by an element variable
(Σ₀ ≡ Σ₁) ⊦ (Γ₀ ⊦ A₀) ≡ (Γ₁ ⊦ A₁) entry

Σ₀ ≡ Σ₁ ⊦ E₀ ≡ E₁ entry
σ₀ ≡ σ₁ : Ξ₀ ≡ Ξ₁ ⇒ Σ₀ ≡ Σ₁
-----------------------------------------  // Signature substitution on entry
Ξ₀ ≡ Ξ₁ ⊦ E₀(σ₀) ≡ E₁(σ₁) entry, where

       (type)(σ) = type     // by induction on syntax of signature entry
       (A)(σ) = A(σ)

///////// Inference rules (sig entry instance) //////////

(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ A₀ ≡ A₁ type
---------------------------------------------
Σ₀ ≡ Σ₁ ⊦ A₀ ≡ A₁ : (Γ₀ ⊦ type) ≡ (Γ₁ ⊦ type)

(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ a₀ ≡ a₁ : A₀ ≡ A₁
-----------------------------------------
Σ₀ ≡ Σ₁ ⊦ a₀ ≡ a₁ : (Γ₀ ⊦ A₀) ≡ (Γ₁ ⊦ A₁)

Σ₀ ≡ Σ₁ ⊦ e₀ ≡ e₁ : E₀ ≡ E₁
σ₀ ≡ σ₁ : Ξ₀ ≡ Ξ₁ ⇒ Σ₀ ≡ Σ₁
---------------------------------------
Σ₀ ≡ Σ₁ ⊦ e₀(σ₀) ≡ e₁(σ₁) : E₀(σ₀) ≡ E₁(σ₁), where

     (a)(σ) = a(σ)  // by induction on syntax of signature entry instance
     (A)(σ) = A(σ)

///////// Inference rules (sig substitution) //////////

· ≡ · : Σ₀ ≡ Σ₁ ⇒ ε ≡ ε

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
ϴ₀ ≡ ϴ₁ ⊦ E₀ ≡ E₁ entry
Σ₀ ≡ Σ₁ ⊦ e₀ ≡ e₁ : E₀(σ₀) ≡ E₁(σ₁)
---------------------------------------------
(σ₀, e₀) ≡ (σ₁, e₁) : Σ₀ ≡ Σ₁ ⇒ ϴ₀ E₀ ≡ ϴ₁ E₁

τ₀ ≡ τ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
σ₀ ≡ σ₁ : ϴ₀ ≡ ϴ₁ ⇒ Ξ₀ ≡ Ξ₁
-----------------------------------------
(σ₀ ∘ τ₀) ≡ (σ₁ ∘ τ₁) : Σ₀ ≡ Σ₁ ⇒ Ξ₀ ≡ Ξ₁

Σ₀ ≡ Σ₁ ⊦ E₀ ≡ Σ₁ entry
-------------------------------
↑ ≡ ↑ : Σ₀ E₀ ≡ Σ₁ E₁ ⇒ Σ₀ ≡ Σ₁

Σ₀ ≡ Σ₁ sig
---------------------------
id ≡ id : Σ₀ ≡ Σ₁ ⇒ Σ₀ ≡ Σ₁

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ε ≡ ε
-------------------------
σ₀ ≡ · : Σ₀ ≡ Σ₁ ⇒ ε ≡ ε

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ε ≡ ε
-------------------------
σ₁ ≡ · : Σ₀ ≡ Σ₁ ⇒ ε ≡ ε

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
---------------------------
id ∘ σ₀ ≡ σ₀ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
---------------------------
id ∘ σ₁ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
---------------------------
σ₀ ∘ id ≡ σ₀ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
--------------------------------
σ₁ ∘ id ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁

σ₁₀ : Σ₁ ⇒ Σ₀
σ₂₁ : Σ₂ ⇒ Σ₁
σ₃₂ : Σ₃ ⇒ Σ₂
-----------------------------------------------
σ₁₀ ∘ (σ₂₁ ∘ σ₃₂) ≡ (σ₁₀ ∘ σ₂₁) ∘ σ₃₂ : Σ₃ ⇒ Σ₀

σ : Σ ⇒ ϴ
ϴ ⊦ E entry
Σ ⊦ e : E(σ)
----------------------------
↑ ∘ (σ, e) ≡ σ : Σ ⇒ ϴ

τ : Σ₀ ⇒ Σ₁
σ : Σ₁ ⇒ Σ₂
Σ₂ ⊦ E entry
Σ₁ ⊦ e : E(σ)
--------------------------------------------
(σ, e) ∘ τ ≡ (σ ∘ τ, e(τ)) : Σ₀ ⇒ Σ₂ E


///////// Inference rules (ctx) //////////

Σ₀ ≡ Σ₁ sig
-------------------
Σ₀ ≡ Σ₁ ⊦ ε ≡ ε ctx

(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ A₀ ≡ A₁ type
----------------------------------
Σ₀ ≡ Σ₁ ⊦ Γ₀ A₀ ≡ Γ₁ A₁ ctx

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
ϴ₀ ≡ ϴ₁ ⊦ Γ₀ ≡ Γ₁ ctx
------------------------------------
Σ₀ ≡ Σ₁ ⊦ Γ₀(σ₀) ≡ Γ₁(σ₁) ctx, where

     ε(σ) ≔ ε                  // by induction on syntax of context
     (Γ A)(σ) ≔ Γ(σ) A(σ)

///////// Inference rules (ctx substitution) //////////

Σ₀ ≡ Σ₁ ⊦ Γ₀ ≡ Γ₁ ctx
---------------------------------
Σ₀ ≡ Σ₁ ⊦ · ≡ · : Γ₀ ≡ Γ₁ ⇒ ε ≡ ε

Σ₀ ≡ Σ₁ ⊦ Γ₀ ≡ Γ₁ ctx
--------------------------------
Σ₀ ≡ Σ₁ ⊦ id : Γ₀ ≡ Γ₁ ⇒ Γ₀ ≡ Γ₁

Σ₀ ≡ Σ₁ ⊦ Γ₀ A₀ ≡ Γ₁ A₁ ctx
-----------------------------------------
Σ₀ ≡ Σ₁ ⊦ ↑ ≡ ↑ : Γ₀ A₀ ≡ Γ₁ A₁ ⇒ Γ₀ ≡ Γ₁

Σ₀ ≡ Σ₁ ⊦ σ₀ ≡ σ₁ : Δ₀ ≡ Δ₁ ⇒ Γ₀ ≡ Γ₁
Σ₀ ≡ Σ₁ ⊦ τ₀ ≡ τ₁ : Ξ₀ ≡ Ξ₁ ⇒ Δ₀ ≡ Δ₁
-------------------------------------------
Σ₀ ≡ Σ₁ ⊦ σ₀ ∘ τ₀ ≡ σ₁ ∘ τ₁ ≡ : Ξ₀ ≡ Ξ₁ ⇒ Γ₀ ≡ Γ₁

Σ₀ ≡ Σ₁ ⊦ σ₀ ≡ σ₁ : Δ₀ ≡ Δ₁ ⇒ Γ₀ ≡ Γ₁
(Σ₀ ≡ Σ₁) (Γ₀ ≡ Γ₁) ⊦ A₀ ≡ A₁ type
(Σ₀ ≡ Σ₁) (Δ₀ ≡ Γ₁) ⊦ t₀ ≡ t₁ : A₀(σ₀) ≡ A₁(σ₁)
---------------------------------------------------
Σ₀ ≡ Σ₁ ⊦ σ₀, t₀ ≡ σ₁, t₁ : Δ₀ ≡ Δ₁ ⇒ Γ₀ A₀ ≡ Γ₁ A₁

σ₀ ≡ σ₁ : Σ₀ ≡ Σ₁ ⇒ ϴ₀ ≡ ϴ₁
ϴ₀ ≡ ϴ₁ ⊦ τ₀ ≡ τ₁ : Γ₀ ≡ Γ₁ ⇒ Δ₀ ≡ Δ₁
-------------------------------------------------------------
Σ₀ ≡ Σ₁ ⊦ τ₀(σ₀) ≡ τ₁(σ₁) : Γ₀(σ₀) ≡ Γ₁(σ₁) ⇒ Δ₀(σ₀) ≡ Δ₁(σ₁)

Σ ⊦ Γ ctx
Σ ⊦ σ : Γ ⇒ ε
-----------------

----------------------

Σ ⊦ σ : Γ ⇒ Δ
----------------------
Σ ⊦ id ∘ σ = σ : Γ ⇒ Δ

Σ ⊦ σ₁₀ : Γ₁ ⇒ Γ₀
Σ ⊦ σ₂₁ : Γ₂ ⇒ Γ₁
Σ ⊦ σ₃₂ : Γ₃ ⇒ Γ₂
---------------------------------------------------
Σ ⊦ σ₁₀ ∘ (σ₂₁ ∘ σ₃₂) = (σ₁₀ ∘ σ₂₁) ∘ σ₃₂ : Γ₃ ⇒ Γ₀

Σ ⊦ σ : Δ ⇒ Γ
Σ Γ ⊦ A type
Σ Δ ⊦ t : A(σ)
--------------------------
Σ ⊦ ↑ ∘ (σ, t) = σ : Δ ⇒ Γ

Σ ⊦ σ : Γ₁ ⇒ Γ₂
