// TypingRule — syntax of derivation steps in .rules files
R ::=
    Γ ctx
  | Γ = Γ' ctx [via Γ'']
  | Γ = Γ' ⊦ t : T
  | Γ | α ctx
  | Γ | α ⊦ T | α type
  | Γ | α ⊦ t | α : T | α type
  | Γ ⊦ σ sub-wf [to Δ [via Θ]]
  | Γ ⊦ T type
  | Γ ⊦ T = T' type [via T'']
  | Γ ⊦ Refl : t{≥5} ∈ T
  | Γ ⊦ (t : T ⨯ T') .π₁
  | Γ ⊦ (t : T ⨯ T') .π₂
  | Γ ⊦ (t : T → T') t{≥5}
  | Γ ⊦ x ≔ t : T
  | Γ ⊦ t = t' : T [via t'' | ↝ T' | from Γ']
  | Γ ⊦ t[σ] : T [from Γ']
  | Γ ⊦ t : T ↝ T'
  | Γ ⊦ t : T reflect
  | Γ ⊦ t : T
  | Γ ⊦ t
