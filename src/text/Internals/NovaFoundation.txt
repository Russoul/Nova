////////// Preface //////////

This file contains a formalisation of Nova's type theory: Extensional Type Theory.
We present the theory extrinsically: https://ncatlab.org/nlab/show/intrinsic+and+extrinsic+views+of+typing

//////////// Type Theory /////////////

We have:
 * Empty type: 𝟘
 * Unit type: 𝟙
 * Natural numbers type: ℕ
 * Dependent product type: Π
 * Dependent sum type: Σ
 * Equality type: (≡) with reflection
 * A universe: 𝕌


/////////////// Notes /////////////////

* We don't define definitional equality relation and then quotient types by it.
   Instead we simply reuse extensional, proof-irrelevant equality of the meta-theory

* Substituion is algorithmic: there is a program that reduces out all substitutions in a type/term/telescope/list/etc.

* Equality is undecidable. Hence "type-checking" is likewise so.

///////////// Syntax ///////////////

// typing context
Γ ::= ε    // empty typing context
    | Γ T  // extended typing context

// typing context substitution
σ, τ ::=
       ·             // unique substitution into the empty context
     | σ, t          // substitution into an extended context
     | σ ∘ σ         // composition of substitutions
     | id            // identity substitution
     | ↑             // weakening substitution

// element
t ::= t(σ)           // (Typing) context substitution
    | λ t            // dependent product type introduction
    | t t            // dependent product type elimination
    | t .π₁          // dependent sum type elimination (1)
    | t .π₂          // dependent sum type elimination (2)
    | t , t          // dependent sum type introduction
    | t → t          // element of the universe (encodes Π)
    | t ⨯ t          // element of the universe (encodes Σ)
    | ℕ              // element of the universe
    | 𝟘              // element of the universe
    | 𝟙              // element of the universe
    | t ≡ t ∈ t      // element of the universe
    | ()             // the only element of 𝟙
    | Z
    | S t
    | ℕ-elim T t t t
    | 𝟘-elim t
    | ☐ᵢ             // i-th element in the typing context
    | Refl           // reflexive equality proof

// type
T ::= 𝕌              // universe type
    | T → T          // dependent product type (Π)
    | T ⨯ T          // dependent sum type (Σ)
    | t ≡ t ∈ T      // extensional equality type
    | ℕ
    | 𝟘
    | 𝟙
    | El t           // Every element of the universe is a type
    | T(σ)           // (Typing) context substitution

// (type) telescope
Δ ::= ε | T Δ

// (element) list
ē ::= · | e ē

σ = σ' : Γ₀ ⇒ Γ₁
--------------------------
Γ₀ ⊦ t(σ) = t'(σ') : A(σ)

Γ A ⊦ f = f' : B
--------------------------
Γ ⊦ λ f = λ f' : A → B

Γ ⊦ f = f' : A → B
Γ ⊦ e = e' : A
--------------------------
Γ ⊦ f e = f' e' : B(id, e)

Γ ⊦ a = a' : A
Γ ⊦ b = b' : B(id, a)
--------------------------
Γ ⊦ a, b = a', b' : A ⨯ B

Γ ⊦ t = t' : A ⨯ B
--------------------------
Γ ⊦ t .π₁ = t' .π₁ : A

Γ ⊦ t = t' : A ⨯ B
-------------------------------------------
Γ ⊦ t .π₂ = t' .π₂ : B(id, t .π₁)

Γ ⊦ t = t' : ℕ
--------------------------
Γ ⊦ S t = S t' : ℕ

Γ ℕ ⊦ A = A' type
Γ ⊦ z = z' : A(id, Z)
Γ ℕ A ⊦ s = s' : A(↑, S ☐₀)
Γ ⊦ t = t' : ℕ
-------------------------------------------
Γ ⊦ ℕ-elim A z s t = ℕ-elim A' z' s' t' : A(id, t)

Γ ⊦ A type
Γ ⊦ t = t' : 𝟘
--------------------------
Γ ⊦ 𝟘-elim t = 𝟘-elim t' : A

// Congruence rules for universe code constructors

Γ ⊦ a = a' : 𝕌
Γ (El a) ⊦ b = b' : 𝕌
--------------------------
Γ ⊦ a → b = a' → b' : 𝕌

Γ ⊦ a = a' : 𝕌
Γ (El a) ⊦ b = b' : 𝕌
--------------------------
Γ ⊦ a ⨯ b = a' ⨯ b' : 𝕌

Γ ⊦ a₀ = a₀' : El A
Γ ⊦ a₁ = a₁' : El A
Γ ⊦ A = A' : 𝕌
-------------------------------------------
Γ ⊦ a₀ ≡ a₁ ∈ A = a₀' ≡ a₁' ∈ A' : 𝕌

// Injectivity of successor

Γ ⊦ S t = S t' : ℕ
--------------------------
Γ ⊦ t = t' : ℕ

// Injectivity of universe code constructors

Γ ⊦ a → b = a' → b' : 𝕌
--------------------------
Γ ⊦ a = a' : 𝕌

Γ ⊦ a → b = a' → b' : 𝕌
--------------------------
Γ (El a) ⊦ b = b' : 𝕌

Γ ⊦ a ⨯ b = a' ⨯ b' : 𝕌
--------------------------
Γ ⊦ a = a' : 𝕌

Γ ⊦ a ⨯ b = a' ⨯ b' : 𝕌
--------------------------
Γ (El a) ⊦ b = b' : 𝕌

Γ ⊦ a₀ ≡ a₁ ∈ A = a₀' ≡ a₁' ∈ A' : 𝕌
-----------------------------------------
Γ ⊦ A = A' : 𝕌

Γ ⊦ a₀ ≡ a₁ ∈ A = a₀' ≡ a₁' ∈ A' : 𝕌
-----------------------------------------
Γ ⊦ a₀ = a₀' : El A

Γ ⊦ a₀ ≡ a₁ ∈ A = a₀' ≡ a₁' ∈ A' : 𝕌
-----------------------------------------
Γ ⊦ a₁ = a₁' : El A


///////// Inference rules (elem list) //////////

Γ ⊦ · : ε

Γ ⊦ Δ tel
Γ ⊦ e : A
Γ ⊦ ē : Δ(id, e)
------------------
Γ ⊦ e ē : A Δ

Γ₁ ⊦ Δ tel
Γ₁ ⊦ ē : Δ
σ : Γ₀ ⇒ Γ₁
--------------------------
Γ₀ ⊦ ē(σ) : Δ(σ)

σ : Γ₀ ⇒ Γ₁
--------------------------
Γ₀ ⊦ ·(σ) = · : ε

Γ₁ ⊦ e ē : A Δ
σ : Γ₀ ⇒ Γ₁
----------------------------------------------
Γ₀ ⊦ (e ē)(σ) = e(σ) ē(σ) : A(σ) Δ(σ⁺)

Γ ⊦ · = · : ε

Γ ⊦ e₀ = e₁ : A
Γ ⊦ ē₀ = ē₁ : Δ(id, e₀)
-------------------------
Γ ⊦ e₀ ē₀ = e₁ ē₁ : A Δ

// Structural rules for element list equality

Γ ⊦ ē : Δ
--------------------
Γ ⊦ ē = ē : Δ

Γ ⊦ ē = ē' : Δ
--------------------
Γ ⊦ ē' = ē : Δ

Γ ⊦ ē = ē' : Δ
Γ ⊦ ē' = ē'' : Δ
--------------------
Γ ⊦ ē = ē'' : Δ

// Conversion rules for element list

Γ ⊦ ē : Δ
Γ = Γ' ctx
-----------
Γ' ⊦ ē : Δ

Γ ⊦ ē : Δ
Γ ⊦ Δ = Δ' tel
--------------
Γ ⊦ ē : Δ'

Γ ⊦ ē = ē' : Δ
Γ = Γ' ctx
-----------
Γ' ⊦ ē = ē' : Δ

Γ ⊦ ē = ē' : Δ
Γ ⊦ Δ = Δ' tel
--------------
Γ ⊦ ē = ē' : Δ'

// Congruence rules for element list constructors

Γ ⊦ e = e' : A
Γ ⊦ ē = ē' : Δ(id, e)
--------------------------
Γ ⊦ e ē = e' ē' : A Δ

Γ₁ ⊦ ē = ē' : Δ
σ = σ' : Γ₀ ⇒ Γ₁
--------------------------
Γ₀ ⊦ ē(σ) = ē'(σ') : Δ(σ)

// Injectivity of non-empty list equality

Γ ⊦ e ē = e' ē' : A Δ
--------------------------
Γ ⊦ e = e' : A

Γ ⊦ e ē = e' ē' : A Δ
--------------------------
Γ ⊦ ē = ē' : Δ(id, e)
