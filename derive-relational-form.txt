// Reflection

Γ ⊦ 𝔽 poly
Γ ⊦ a : 𝕌
--------------
Γ ⊦ ⌊𝔽⌋(a) : 𝕌
    ⌊𝕏⌋(a) ≜ a
    ⌊K b⌋(a) ≜ b
    ⌊𝔽 ⊎ 𝔾⌋(a) ≜ ⌊𝔽⌋(a) ⊎ ⌊𝔾⌋(a)
    ⌊𝔽 ⨯ 𝔾⌋(a) ≜ ⌊𝔽⌋(a) ⨯ ⌊𝔾⌋(a)
    ⌊(x : a) ⨯ 𝔽⌋(a) ≜ (x : a) ⨯ ⌊𝔽⌋(a)
    ⌊(x : a) → 𝔽⌋(a) ≜ (x : a) → ⌊𝔽⌋(a)

Γ ⊦ 𝔽 poly
Γ ⊦ a : 𝕌
Γ ⊦ b : 𝕌
Γ ⊦ f : a → b
---------------------------
Γ ⊦ map_𝔽 f : ⌊𝔽⌋(a) → ⌊𝔽⌋(b)
    map_𝕏 f ≜ f : a → b
    map_{K c} f ≜ id : c → c
    map_{𝔽 ⊎ 𝔾} f ≜ λx. ⊎-elim (x. inj₁ (map_𝔽 f x)) (x. inj₂ (map_𝔾 f x)) x : ⌊𝔽⌋(a) ⊎ ⌊𝔾⌋(a) → ⌊𝔽⌋(b) ⊎ ⌊𝔾⌋(b)
    map_{𝔽 ⨯ 𝔾} f ≜ λx. map_𝔽 f (x .π₁), map_𝔾 f (x. π₂) ⌊𝔽⌋(a) ⨯ ⌊𝔾⌋(a) → ⌊𝔽⌋(b) ⨯ ⌊𝔾⌋(b)
    map_{(t : c) ⨯ 𝔽} f ≜ λx. (x. π₁, map_{𝔽[x .π₁ / t]} f (x .π₂)) : ((t : c) ⨯ ⌊𝔽⌋(a)) → ((t : c) ⨯ ⌊𝔽⌋(b))
    map_{(t : c) → 𝔽} f ≜ λα. λt. map_𝔽 f (α t) : ((t : c) → ⌊𝔽⌋(a)) → ((t : c) → ⌊𝔽⌋(b))
