i, j, k ::= <variable> // interval variable

x, y, z ::= <variable> // kan variable

Ξ ::= ε | Ξ i // cube

r ::= 0 | 1 | i // interval element

α, φ, ψ ::= ⊤ | ⊥ | r = r | δ r | φ ∨ φ | φ ∧ φ | ∀i. φ

A, B, T ::= (x : T) → T   // kan type
          | (x : T) ⨯ T
          | Path x.T t t
          | t ≡ t ∈ T
          | Bool
          | S¹
          | V r T T t
          | 𝕌
          | Contr T
          | Equiv T T

Γ ::= ε | Γ (x : T)          // kan context

M, N, P, t, f, e ::=         // kan element
                  x ↦ t
                | t t
                | (t, t)
                | t .π₁
                | t .π₂
                | ⟨i⟩ t
                | t @ r
                | True
                | False
                | if x.T t t t
                | Base
                | Loop r
                | S¹-elim x.T t i.t
                | coe r r i.T t
                | hcom r r φ T t i.t
                | com r r φ i.T t i.t
                | V r T T t
                | Vin r t t
                | Vproj r t
                | Glue α T T t
                | glue α t t
                | unglue t
                | [φ ↪ t | φ ↪ t]
                | []
