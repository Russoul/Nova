SigWf : Sig -> Ω
SigWfEps : SigWf ε

CtxWf : (Σ : Sig) -> Ctx -> SigWf Σ -> Ω
TyWf : (Σ : Sig) (Γ : Ctx) (A : Ty) -> (α : SigWf Σ) -> CtxWf Σ Γ α -> Ω
ElemWf : (Σ : Sig) (Γ : Ctx) (A : Ty) (a : Elem)
      -> (α : SigWf Σ) (β : CtxWf Σ Γ α) -> TyWf Σ Γ A α β -> Ω
SubWf : (Σ : Sig) (Γ : Ctx) (Δ : Ctx) (σ : Sub) -> (α : SigWf Σ) -> CtxWf Σ Γ α -> CtxWf Σ Δ α -> Ω
SubNormWf : (Σ : Sig) (Γ : Ctx) (Δ : Ctx) (σ : SubNorm) -> (α : SigWf Σ) -> CtxWf Σ Γ α -> CtxWf Σ Δ α -> Ω
SubNormEq : (Σ : Sig) (Γ : Ctx) (Δ : Ctx) (σ₀ σ₁ : SubNorm)
         -> (α : SigWf Σ) -> (β : CtxWf Σ Γ α) -> (γ : CtxWf Σ Δ α)
         -> SubNormWf Σ Γ Δ σ₀ α β γ
         -> SubNormWf Σ Γ Δ σ₁ α β γ
         -> Ω
TyEq : (Σ : Sig) (Γ : Ctx) (A₀ A₁ : Ty) -> (α : SigWf Σ) -> (β : CtxWf Σ Γ α)
    -> TyWf Σ Γ A₀ α β
    -> TyWf Σ Γ A₁ α β
    -> Ω
ElemEq : (Σ : Sig) (Γ : Ctx) (A : Ty) (a₀ a₁ : Elem)
      -> (α : SigWf Σ) (β : CtxWf Σ Γ α) (γ : TyWf Σ Γ A α β)
      -> ElemWf Σ Γ A a₀ α β γ
      -> ElemWf Σ Γ A a₁ α β γ
      -> Ω
Fresh : Sig -> Identifier -> Ω


ElabGamma : SurfaceCtx -> (α : ⟦Σ sig⟧) -> ⟦Σ ⊦ Γ ctx⟧ α -> Ω
ElabGammaEps : (α : ⟦Σ sig⟧) -> ElabGamma ε α ⟦Σ ⊦ ε ctx⟧ α
ElabGammaExt : (α : ⟦Σ sig⟧) (β : ⟦Σ ⊦ Γ ctx⟧ α) (γ : ⟦Σ Γ ⊦ A type⟧ α β)
            -> ElabGamma gam α β
            -> ElabTy ty α β γ
            -> ElabGamma ⟦gam ᐅ ty⟧ α (⟦Σ ⊦ Γ ᐅ A ctx⟧ α (β ᐅ γ))

ElabGammaEq : SurfaceCtxEq -> (α : SigWf Σ) -> CtxWf Σ Γ₀ α -> CtxWf Σ Γ₁ α -> Ω
ElabSub : (Σ : Sig) (Γ : Ctx) (Δ : Ctx) (σ : Sub) (s : SurfaceSub)
       -> (α : SigWf Σ) (β : CtxWf Σ Γ α) (γ : CtxWf Σ Δ α) -> SubWf Σ Γ Δ σ α β γ ->
       -> Ω
ElabSubNorm : (Σ : Sig) (Γ : Ctx) (Δ : Ctx) (σ : SubNorm) (s : SurfaceSubNorm)
           -> (α : SigWf Σ) (β : CtxWf Σ Γ α) (γ : CtxWf Σ Δ α) -> SubNormWf Σ Γ Δ σ α β γ ->
           -> Ω
ElabSubNormEq : (Σ : Sig) (Γ : Ctx) (Δ : Ctx) (σ₀ σ₁ : SubNorm) (s : SurfaceSubNormEq)
             -> (α : SigWf Σ) (β : CtxWf Σ Γ α) (γ : CtxWf Σ Δ α)
             -> SubNormWf Σ Γ Δ σ₀ α β γ
             -> SubNormWf Σ Γ Δ σ₁ α β γ
             -> Ω
ElabTy : SurfaceTy
      -> (α : SigWf Σ) (β : CtxWf Σ Γ α)
      -> TyWf Σ Γ A α β
      -> Ω
ElabTyEq : (Σ : Sig) (Γ : Ctx) (A₀ A₁ : Ty) (s : SurfaceTyEq)
        -> (α : SigWf Σ) (β : CtxWf Σ Γ α)
        -> TyWf Σ Γ A₀ α β
        -> TyWf Σ Γ A₁ α β
        -> Ω
ElabElem : (Σ : Sig) (Γ : Ctx) (A : Ty) (a : Elem) (s : SurfaceElem)
        -> (α : SigWf Σ) (β : CtxWf Σ Γ α) (γ : TyWf Σ Γ A α β)
        -> ElemWf Σ Γ A a α β γ
        -> Ω
ElabElemEq : (Σ : Sig) (Γ : Ctx) (A : Ty) (a₀ a₁ : Elem) (s : SurfaceElemEq)
          -> (α : SigWf Σ) (β : CtxWf Σ Γ α) (γ : TyWf Σ Γ A α β)
          -> ElemWf Σ Γ A a₀ α β γ -> ElemWf Σ Γ A a₁ α β γ
          -> Ω
ElabSig : (Σ : Sig) -> SurfaceSig -> SigWf Σ -> Ω
ElabEntry : (Σ : Sig) (Γ : Ctx) (A : Ty) (a : Elem) (x : Identifier)
         -> SurfaceEntry
         -> (α : SigWf Σ)
         -> (β : CtxWf Σ Γ α)
         -> (γ : TyWf Σ Γ A α β)
         -> ElemWf Σ Γ A a α β γ
         -> Fresh Σ x
         -> Ω

/////////////// Rules ///////////////

Σ sig
Σ ⊦ Γ ctx
--------------------------
Σ ⊦ Γ = ε ctx
==========================
Σ ⊦ ε ~> Γ ctx

Σ sig
Σ ⊦ ?Γ ctx
--------------------------
Σ ⊦ ?Δ ctx
Σ ?Δ ⊦ ?A type
Σ ⊦ ?Γ = ?Δ ᐅ ?A
Σ ⊦ Δ ~> ?Δ ctx
Σ ?Δ ⊦ A ~> ?A type
==========================
Σ ⊦ Δ ᐅ A ~> ?Γ ctx


Σ sig
Σ ⊦ Γ₀ ctx
Σ ⊦ Γ₁ ctx
--------------------------
Σ ⊦ Γ₀ = Γ₁ ctx
==========================
Σ ⊦ refl shows Γ₀ ≐ Γ₁ ctx

Σ sig
--------------------------
⊤
==========================
Σ ⊦ ε shows ε ≐ ε ctx

Σ sig
Σ ⊦ Γ₀ ctx
Σ ⊦ Γ₁ ctx
---------------------------
Σ ⊦ Γ⁼ shows Γ₁ ≐ Γ₀ ctx
===========================
Σ ⊦ Γ⁼ ⁻¹ shows Γ₀ ≐ Γ₁ ctx

Σ sig
Σ ⊦ Γ₀ ctx
Σ ⊦ Γ₂ ctx
-------------------------------
Σ ⊦ Γ₁ ctx
Σ ⊦ Γ⁼₀ shows Γ₀ ≐ Γ₁ ctx
Σ ⊦ Γ⁼₂ shows Γ₁ ≐ Γ₂ ctx
===============================
Σ ⊦ Γ⁼₀ · Γ⁼₂ shows Γ₀ ≐ Γ₂ ctx

Σ sig
Σ ⊦ Γ₀ ctx
Σ ⊦ Γ₁ ctx
----------------------------
Σ ⊦ Δ₀ ctx
Σ Δ₀ ⊦ A₀ type
Σ ⊦ Γ₀ = Δ₀ ᐅ A₀ ctx
Σ ⊦ Δ₁ ctx
Σ Δ₁ ⊦ A₁ type
Σ ⊦ Γ₁ = Δ₁ ᐅ A₁ ctx
Σ ⊦ Δ⁼ shows Δ₀ ≐ Δ₁ ctx
Σ Δ₁ ⊦ A⁼ shows A₀ ≐ A₁ type
=============================
Σ ⊦ Δ⁼ ᐅ A⁼ shows Γ₀ ≐ Γ₁ ctx

Σ Γ ⊦ A = Γ‖ₙ type

Σ Γ ⊦ ☐ₙ ~> ☐ₙ : A

Σ Γ ⊦ Z ~> Z : A

Σ sig
Σ ⊦ Γ ctx
Σ Γ ⊦ T type
----------------
Σ Γ ⊦ A type
Σ Γ ᐅ A ⊦ B type
Σ Γ ᐅ A ⊦ f ~> ?f : B
Σ Γ ⊦ T = A → B type
----------------------
Σ Γ ⊦ λ f ~> λ ?f : T

Σ sig
Σ ⊦ Γ ctx
Σ Γ ⊦ T type
------------------------
Σ Γ ⊦ A type
Σ Γ ᐅ A ⊦ B type
Σ Γ ⊦ f ~> ?f : A → B
Σ Γ ⊦ e ~> ?e : A
Σ Γ ⊦ T = B[id, e] type
------------------------
Σ Γ ⊦ f e ~> ?f ?e : T

Σ sig
Σ ⊦ Γ ctx
Σ Γ ⊦ T type

Σ Γ ⊦ t : T
================
Σ Γ ⊦ _ ~> ?t : T

============================

=================
Σ Γ ⊦ t : T
Σ Γ ⊦ t = f e : T
