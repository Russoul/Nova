σ{2} ::= · | id | ↑ | (σ{≥0})
σ{1} ::= σ{≥2} ∘ σ{≥1}                                                      Chain (right-assoc)
σ{0} ::= σ{≥0} , t{≥1}                                                      Ext (left-assoc)

T{4} ::= 𝟘 | 𝟙 | ℕ | 𝕌 | (T{≥0})
T{3} ::= T{≥3}[σ{≥0}] // FIXME: doesn't exist anymore
T{2} ::= El t{≥4}
T{1} ::= T{≥2} → T{≥1} | T{≥2} ⨯ T{≥1}                                      (right-assoc)
T{0} ::= t{≥2} ≡ t{≥2} ∈ T{≥1}

t˲ ::= · | t˲ , t

t{5} ::= ☐ₙ | () | Z | Refl | 𝟘 | 𝟙 | ℕ | x[t˲] | (t{≥0})
t{4} ::= t{≥4} [σ{≥0}] // FIXME: doesn't exist anymore
t{3} ::= t{≥3} t{≥4}  t{≥3} .π₁ | t{≥3} .π₂
t{2} ::= λ t{≥3} | 𝟘-elim t{≥4} | S t{≥4} | ℕ-elim t{≥4} t{≥4} t{≥4}
t{1} ::= t{≥2} → t{≥1} | t{≥2} ⨯ t{≥1} | t{≥2} ≡ t{≥2} ∈ t{≥2}
t{0} ::= t{≥1} , t{≥0}                                                      SigmaIntro (right-assoc)

Γ ::= ε | Γ ᐅ T{≥0}                                                          (left-assoc)
Δ ::= ε | T{≥0} ◁ Δ                                                          (right-assoc)
ē ::= · | t{≥1} (, t{≥1})*

α{5} ::= ↓ | id | (α{≥0})
α{4} ::= α{≥4} [α{≥0}] // FIXME: doesn't exist anymore
α{3} ::= α{≥3} α{≥4} | α{≥3} .π₁ | α{≥3} .π₂                               (left-assoc)
α{2} ::= λ α{≥4} | 𝟘-elim α{≥4} | S α{≥4} | El α{≥4} | ℕ-elim α{≥4} α{≥4} α{≥4}
α{1} ::= α{≥2} → α{≥1} | α{≥2} ⨯ α{≥1} | α{≥2} ᐅ α{≥1}                     (right-assoc)
       | α{≥2} ≡ α{≥3} ∈ α{≥3}
α{0} ::= α{≥1} , α{≥0} | α{≥1} ; α{≥0}                                       (right-assoc)

// TypingRule — syntax of derivation steps in .rules files
// Each rule starts with a unique keyword; arguments follow in conclusion-like notation.
R ::=

  // Context
  ctx-emp                                        (CtxWfEmpty)
  ctx-ext    Γ ᐅ T                               (CtxWfExt)
  ctx-refl   Γ                                   (CtxEqRefl)
  ctx-sym    Γ₁ = Γ₀                             (CtxEqSym, conclusion: Γ₁ = Γ₀)
  ctx-trans  Γ₀ = Γ₂ via Γ₁                     (CtxEqTrans, conclusion: Γ₀ = Γ₂)
  ctx-cmp    Γ via α                              (CtxWfCompute)

  // Substitution wf
  sub-term   Γ ⊦ ·
  sub-id     Γ ⊦ id
  sub-wk     Γ ⊦ ↑
  sub-ext    Γ ⊦ σ, e to Δ ᐅ T
  sub-chn    Γ ⊦ σ ∘ τ to Δ via Θ

  // Substitution eq
  sub-refl   Γ ⊦ σ : Δ
  sub-sym    Γ ⊦ σ₁ = σ₀ : Δ                   (conclusion: σ₁ = σ₀)
  sub-trans  Γ ⊦ σ₀ = σ₂ : Δ via σ₁            (conclusion: σ₀ = σ₂)

  // Normal substitution wf
  sub-norm-term   Γ ⊦ ·                          (SubNormWfTerminal)
  sub-norm-ext    Γ ⊦ e˲, t to Δ ᐅ T             (SubNormWfExt)
  sub-norm-chn    Δ ⊦ e˲ ∘ σ to Γ₁ via Γ₀        (SubNormWfChain)

  // Normal substitution eq
  sub-norm-refl    Γ ⊦ e˲ : Δ
  sub-norm-sym     Γ ⊦ e˲₁ = e˲₀ : Δ                 (conclusion: e˲₁ = e˲₀)
  sub-norm-trans   Γ ⊦ e˲₀ = e˲₂ : Δ via e˲₁         (conclusion: e˲₀ = e˲₂)
  sub-norm-ext-eq  Γ ⊦ e˲₀, t₀ = e˲₁, t₁ : Δ ᐅ T    (SubNormEqExt, ext-eq before ext — longer keyword first)

  // Type wf
  ty-zero    Γ ⊦ 𝟘
  ty-one     Γ ⊦ 𝟙
  ty-nat     Γ ⊦ ℕ
  ty-univ    Γ ⊦ 𝕌
  ty-pi      Γ ⊦ A → B
  ty-sigma   Γ ⊦ A ⨯ B
  ty-eq-form Γ ⊦ l ≡ r ∈ A
  ty-el      Γ ⊦ El e
  ty-sub     Γ ⊦ A[σ] from Δ                    (TyWfSubElim) // FIXME: don't need that anymore
  ty-cmp     Γ via α ⊦ T via β                  (TyWfCompute)

  // Type eq
  ty-refl    Γ ⊦ T
  ty-sym     Γ ⊦ T₁ = T₀                        (conclusion: T₁ = T₀)
  ty-trans   Γ ⊦ T₀ = T₂ via T₁                 (conclusion: T₀ = T₂)

  // Element wf: intro / elim
  el-var      Γ ⊦ ☐ₙ
  el-one      Γ ⊦ ()
  el-zero     Γ ⊦ Z
  el-suc      Γ ⊦ S t
  el-pi-i     Γ ⊦ λ f : A → B
  el-pi-e     Γ ⊦ (f : A → B) e
  el-sigma-i  Γ ⊦ u, v : A ⨯ B
  el-sigma-e1 Γ ⊦ (e : A ⨯ B) .π₁
  el-sigma-e2 Γ ⊦ (e : A ⨯ B) .π₂
  el-zero-e   Γ ⊦ 𝟘-elim e : T
  el-nat-e    Γ ⊦ ℕ-elim z s t : T
  el-reflect  Γ ⊦ t : (l ≡ r ∈ A) reflect       (ElemEqReflection)
  el-refl     Γ ⊦ Refl : t ∈ A                  (ElemWfRefl)
  el-sub      Γ ⊦ t[σ] : T from Δ     // FIXME: don't need that anymore
  el-ty-coe   Γ ⊦ t : T₀ ↝ T₁
  el-ctx-coe  Γ₀ = Γ₁ ⊦ t : T

  // Element wf: universe codes
  el-zero-ty  Γ ⊦ 𝟘 : 𝕌
  el-one-ty   Γ ⊦ 𝟙 : 𝕌
  el-nat-ty   Γ ⊦ ℕ : 𝕌
  el-pi-ty    Γ ⊦ A → B : 𝕌
  el-sigma-ty Γ ⊦ A ⨯ B : 𝕌
  el-eq-ty    Γ ⊦ l ≡ r ∈ A : 𝕌
  el-cmp      Γ via α ⊦ t via β : T via γ       (ElemWfCompute)

  // Signature
  sig-var-eq Γ ⊦ x[e˲]                           (ElemEqSigVar)
  sig-var    Γ ⊦ x[e˲]                           (ElemWfSigVar)
  sig        Γ ⊦ x ≔ t : T                      (SigExt)

  // Element equality
  el-ty-coe-eq Γ ⊦ t₀ = t₁ : T₀ ↝ T₁           (ElemEqTyCoe)
  el-eq-refl   Γ ⊦ t : T                        (ElemEqRefl)
  el-eq-sym    Γ ⊦ t₁ = t₀ : T                  (conclusion: t₁ = t₀)
  el-eq-trans  Γ ⊦ t₀ = t₂ : T via t₁           (conclusion: t₀ = t₂)
  el-sub-cong  Δ ⊦ a[σ] = b[σ] : T from Γ       (ElemEqSubstCong) //FIXME: don't need that anymore as t[σ] computes

  // Telescope equality
  tel-refl   Γ ⊦ Δ
  tel-sym    Γ ⊦ Δ₁ = Δ₀
  tel-trans  Γ ⊦ Δ₀ = Δ₂ via Δ₁

  // Spine equality
  sp-refl    Γ ⊦ s : Δ
  sp-sym     Γ ⊦ s₁ = s₀ : Δ
  sp-trans   Γ ⊦ s₀ = s₂ : Δ via s₁

// JudgementForm — syntax of target goals in .target files
// Each form starts with a unique keyword; no trailing disambiguators needed.
J ::=
  ctx-wf  Γ                           (JfCtxWf)
  ctx-eq  Γ = Γ'                      (JfCtxEq)
  sub-wf  σ : Γ ⇒ Δ                   (JfSubWf)
  sub-eq  σ = σ' : Γ ⇒ Δ              (JfSubEq)
  sub-norm-wf  e˲ : Γ ⇒ Δ norm        (JfSubNormWf)
  sub-norm-eq  e˲ = e˲' : Γ ⇒ Δ norm  (JfSubNormEq)
  ty-wf   Γ ⊦ T                       (JfTyWf)
  ty-eq   Γ ⊦ T = T'                  (JfTyEq)
  el-wf   Γ ⊦ t : T                   (JfElemWf)
  el-eq   Γ ⊦ t = t' : T              (JfElemEq)
  tel-wf  Γ ⊦ Δ                       (JfTelWf)
  tel-eq  Γ ⊦ Δ = Δ'                  (JfTelEq)
  sp-wf   Γ ⊦ ē : Δ                   (JfSpineWf)
  sp-eq   Γ ⊦ ē = ē' : Δ              (JfSpineEq)
